<script>
  // interactive tech dot-grid background
  (function(){
    const canvas = document.getElementById('bgCanvas');
    const ctx = canvas.getContext('2d');
    const dpr = Math.min(window.devicePixelRatio || 1, 2);
    const spacing = 34;
    const radius = 150;
    const radiusSq = radius * radius;

    const reduceMotion = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
    const canHover = window.matchMedia('(hover: hover)').matches;

    let w = 0, h = 0, dots = [];
    const mouse = { x: -9999, y: -9999, active: false };
    let rafId = null;
    let resizeTimer = null;

    function buildDots(){
      dots = [];
      const cols = Math.ceil(w / spacing) + 1;
      const rows = Math.ceil(h / spacing) + 1;
      for(let i = 0; i < cols; i++){
        for(let j = 0; j < rows; j++){
          dots.push({ x: i * spacing, y: j * spacing, glow: 0 });
        }
      }
    }

    function resize(){
      w = window.innerWidth;
      h = window.innerHeight;
      canvas.width = w * dpr;
      canvas.height = h * dpr;
      canvas.style.width = w + 'px';
      canvas.style.height = h + 'px';
      ctx.setTransform(dpr, 0, 0, dpr, 0, 0);
      buildDots();
      renderFrame();
    }

    function renderFrame(){
      ctx.clearRect(0, 0, w, h);

      const near = [];

      for(const d of dots){
        let target = 0;
        if(mouse.active){
          const dx = d.x - mouse.x, dy = d.y - mouse.y;
          const distSq = dx * dx + dy * dy;
          if(distSq < radiusSq){
            target = 1 - Math.sqrt(distSq) / radius;
          }
        }
        d.glow += (target - d.glow) * 0.15;
        if(d.glow > 0.02) near.push(d);

        const alpha = 0.05 + d.glow * 0.9;
        const rad = 1 + d.glow * 2.4;
        ctx.beginPath();
        ctx.fillStyle = 'rgba(255,255,255,' + alpha + ')';
        ctx.arc(d.x, d.y, rad, 0, Math.PI * 2);
        ctx.fill();
      }

      if(near.length > 1){
        ctx.lineWidth = 1;
        const linkDistSq = spacing * spacing * 2.6;
        for(let i = 0; i < near.length; i++){
          for(let j = i + 1; j < near.length; j++){
            const dx = near[i].x - near[j].x, dy = near[i].y - near[j].y;
            const distSq = dx * dx + dy * dy;
            if(distSq < linkDistSq){
              const a = Math.min(near[i].glow, near[j].glow) * 0.5;
              ctx.strokeStyle = 'rgba(255,255,255,' + a + ')';
              ctx.beginPath();
              ctx.moveTo(near[i].x, near[i].y);
              ctx.lineTo(near[j].x, near[j].y);
              ctx.stroke();
            }
          }
        }
      }
    }

    function animate(){
      renderFrame();
      rafId = requestAnimationFrame(animate);
    }

    window.addEventListener('resize', () => {
      clearTimeout(resizeTimer);
      resizeTimer = setTimeout(resize, 120);
    });

    if(canHover && !reduceMotion){
      window.addEventListener('mousemove', (e) => {
        mouse.x = e.clientX;
        mouse.y = e.clientY;
        mouse.active = true;
      });
      window.addEventListener('mouseleave', () => { mouse.active = false; });
      document.addEventListener('visibilitychange', () => {
        if(document.hidden && rafId){ cancelAnimationFrame(rafId); rafId = null; }
        else if(!document.hidden && !rafId){ animate(); }
      });
      resize();
      animate();
    } else {
      resize();
    }
  })();

  // mobile sidebar toggle
  const sidebar = document.getElementById('sidebar');
  const backdrop = document.getElementById('backdrop');
  const menuToggle = document.getElementById('menuToggle');
  function closeSidebar(){ sidebar.classList.remove('open'); backdrop.classList.remove('open'); }
  menuToggle.addEventListener('click', () => {
    sidebar.classList.toggle('open');
    backdrop.classList.toggle('open');
  });
  backdrop.addEventListener('click', closeSidebar);
  sidebar.querySelectorAll('a').forEach(a => a.addEventListener('click', closeSidebar));

  // tabs click -> scroll
  document.querySelectorAll('.tab').forEach(tab => {
    tab.addEventListener('click', () => {
      const el = document.getElementById(tab.dataset.target);
      if(el) el.scrollIntoView({ behavior: 'smooth', block: 'start' });
    });
  });

  // scrollspy
  const sectionIds = ['about','experience','projects','skills','highlights','connect'];
  const sections = sectionIds.map(id => document.getElementById(id)).filter(Boolean);
  const treeLinks = Array.from(document.querySelectorAll('.tree-item[href]'));
  const tabButtons = Array.from(document.querySelectorAll('.tab'));

  function setActive(id){
    treeLinks.forEach(l => l.classList.toggle('active', l.getAttribute('href') === '#' + id));
    tabButtons.forEach(t => t.classList.toggle('active', t.dataset.target === id));
  }

  const observer = new IntersectionObserver((entries) => {
    let best = null;
    entries.forEach(entry => {
      if(entry.isIntersecting){
        if(!best || entry.intersectionRatio > best.intersectionRatio) best = entry;
      }
    });
    if(best) setActive(best.target.id);
  }, { rootMargin: '-44px 0px -55% 0px', threshold: [0.1, 0.25, 0.5, 0.75] });

  sections.forEach(s => observer.observe(s));

  // hero typing sequence
  const reduceMotion = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  const termBody = document.getElementById('termBody');
  const sequence = [
    { type: 'cmd', text: 'cat role.txt' },
    { type: 'out', text: 'Computer Science Engineering Student, focused on backend systems & ML' },
    { type: 'cmd', text: 'echo $STATUS' },
    { type: 'out', text: 'available_for_internships = true' }
  ];

  async function typeLine(text, el){
    for(let i = 0; i <= text.length; i++){
      el.textContent = text.slice(0, i);
      await new Promise(r => setTimeout(r, 16));
    }
  }

  async function runSequence(){
    if(reduceMotion){
      sequence.forEach(step => {
        const line = document.createElement('div');
        line.className = 'term-line';
        if(step.type === 'cmd'){
          line.innerHTML = '<span class="prompt">$</span><span class="type">' + step.text + '</span>';
        } else {
          line.innerHTML = '<span class="out">' + step.text + '</span>';
        }
        termBody.appendChild(line);
      });
      return;
    }
    for(const step of sequence){
      await new Promise(r => setTimeout(r, 260));
      const line = document.createElement('div');
      line.className = 'term-line';
      termBody.appendChild(line);
      if(step.type === 'cmd'){
        line.innerHTML = '<span class="prompt">$</span><span class="type"></span><span class="cursor"></span>';
        const typeEl = line.querySelector('.type');
        const cursorEl = line.querySelector('.cursor');
        await typeLine(step.text, typeEl);
        cursorEl.remove();
      } else {
        line.innerHTML = '<span class="out"></span>';
        const outEl = line.querySelector('.out');
        await typeLine(step.text, outEl);
      }
    }
  }
  runSequence();

  // ===== PIXEL-ART ANIMATED BANNER =====
  (function(){
    const SETS = [
      { folder: 'Animations/Pixel-Art Background 1', count: 141, name: 'Forest Dawn' },
      { folder: 'Animations/Pixel-Art Background 2', count: 153, name: 'Meadow Sunrise' },
      { folder: 'Animations/Pixel-Art Background 3', count: 143, name: 'Night City' },
      { folder: 'Animations/Pixel-Art Background 4', count: 67,  name: 'Cyberpunk Skyline' }
    ];

    const mainCanvas  = document.getElementById('bannerCanvas');
    const fadeCanvas   = document.getElementById('bannerCanvasFade');
    const mainCtx     = mainCanvas.getContext('2d');
    const fadeCtx     = fadeCanvas.getContext('2d');
    const clockEl     = document.getElementById('bannerClock');
    const labelEl     = document.getElementById('sceneLabel');
    const progressEl  = document.getElementById('sceneProgress');
    const bannerEl    = document.getElementById('pixelBanner');

    let setIdx = 0, frameIdx = 0;
    let frames = [], nextFrames = [];
    let playing = true, rafId = null;
    const FPS = 12;
    const INTERVAL = 1000 / FPS;
    let lastTime = 0;

    function pad(n){ return String(n).padStart(4,'0'); }
    function pad2(n){ return String(n).padStart(2,'0'); }

    function loadSet(idx){
      return new Promise(function(resolve){
        var s = SETS[idx], imgs = [], done = 0;
        for(var i = 0; i < s.count; i++){
          (function(j){
            var img = new Image();
            img.src = s.folder + '/frame' + pad(j) + '.png';
            img.onload = img.onerror = function(){
              done++;
              if(done >= s.count) resolve(imgs);
            };
            imgs.push(img);
          })(i);
        }
      });
    }

    function sizeCanvas(){
      var r = bannerEl.getBoundingClientRect();
      mainCanvas.width = r.width;
      mainCanvas.height = r.height;
      fadeCanvas.width = r.width;
      fadeCanvas.height = r.height;
      mainCtx.imageSmoothingEnabled = false;
      fadeCtx.imageSmoothingEnabled = false;
    }

    function drawImg(ctx2d, img){
      if(!img || !img.naturalWidth) return;
      var cw = ctx2d.canvas.width, ch = ctx2d.canvas.height;
      var iw = img.naturalWidth, ih = img.naturalHeight;
      var sc = Math.max(cw/iw, ch/ih);
      var dw = iw*sc, dh = ih*sc;
      ctx2d.clearRect(0,0,cw,ch);
      ctx2d.drawImage(img, (cw-dw)/2, (ch-dh)/2, dw, dh);
    }

    function tick(){
      var now = new Date();
      clockEl.textContent = pad2(now.getHours()) + ':' + pad2(now.getMinutes()) + ':' + pad2(now.getSeconds());
    }

    function updateUI(){
      labelEl.textContent = SETS[setIdx].name + ' \u00b7 scene ' + (setIdx+1) + '/4';
      var pct = ((frameIdx+1)/SETS[setIdx].count)*100;
      progressEl.style.width = pct + '%';
    }

    function crossfade(){
      var next = (setIdx+1) % SETS.length;
      if(!nextFrames.length){
        loadSet(next).then(function(f){
          nextFrames = f;
          doCrossfade(next);
        });
      } else {
        doCrossfade(next);
      }
    }

    function doCrossfade(next){
      drawImg(fadeCtx, nextFrames[0]);
      fadeCanvas.style.opacity = '1';
      setTimeout(function(){
        setIdx = next;
        frameIdx = 0;
        frames = nextFrames;
        nextFrames = [];
        drawImg(mainCtx, frames[0]);
        fadeCanvas.style.opacity = '0';
        updateUI();
        // preload next
        var pn = (setIdx+1) % SETS.length;
        loadSet(pn).then(function(f){ nextFrames = f; });
      }, 1000);
    }

    function animate(ts){
      if(!playing) return;
      rafId = requestAnimationFrame(animate);
      if(ts - lastTime < INTERVAL) return;
      lastTime = ts;
      if(frames.length && frameIdx < frames.length){
        drawImg(mainCtx, frames[frameIdx]);
        updateUI();
        frameIdx++;
        if(frameIdx >= frames.length) crossfade();
      }
    }

    document.addEventListener('visibilitychange', function(){
      if(document.hidden){ playing=false; if(rafId) cancelAnimationFrame(rafId); }
      else { playing=true; lastTime=0; rafId=requestAnimationFrame(animate); }
    });

    tick();
    setInterval(tick, 1000);
    window.addEventListener('resize', sizeCanvas);
    sizeCanvas();

    loadSet(0).then(function(f){
      frames = f;
      frameIdx = 0;
      updateUI();
      playing = true;
      rafId = requestAnimationFrame(animate);
      loadSet(1).then(function(nf){ nextFrames = nf; });
    });
  })();
</script>
